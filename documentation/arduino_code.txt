#include "usb_keyboard.h"

const int BUTTON_COUNT = 8; 
const int LED_COUNT = 8;

const int buttonPin[BUTTON_COUNT]= {2,3,4,5,6,7,8,9};  // the number of the pushbutton pin
const int ledPin[LED_COUNT] = {13,14,15,16,17,18,19,20};    // the number of the LED pin

int lastButtonState[BUTTON_COUNT] = {HIGH, HIGH, HIGH, HIGH, HIGH, HIGH, HIGH, HIGH};
char keyPresses[BUTTON_COUNT] =   {'P','Q','R','T','W','A','F','E'};
int buttonStates[BUTTON_COUNT];

void setup() {

  Serial.begin(9600);
  for (int i = 0; i < BUTTON_COUNT; i++){
    // initialize the pushbutton pin as an input:
    pinMode(buttonPin[i], INPUT);
    // initialize the LED pin as an output:
    pinMode(ledPin[i], OUTPUT);

    // Serial.print("Button");
    // Serial.print(i);
    // Serial.print("on pin");
    // Serial.print(buttonPin[i]);
  }

  
}

void loop() {
  for (int i = 0; i < BUTTON_COUNT; i++){
    buttonStates[i] = digitalRead(buttonPin[i]);
    if (buttonStates[i] != lastButtonState[i]){
      if (buttonStates[i] == HIGH){
        digitalWrite(ledPin[i], HIGH);
        
        //button press
        Keyboard.press(keyPresses[i]);
        delay(50);
        Keyboard.release(keyPresses[i]);
      }
      else{
        digitalWrite(ledPin[i], LOW);
      }
      
      lastButtonState[i]= buttonStates[i];
    }
    
  }
  delay(10);

  
}
